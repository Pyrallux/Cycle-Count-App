Backend:
docker build --secret id=django-private-key,src="../django-private-key.txt" -t pyrallux/ccbb-backend:latest .
docker push pyrallux/ccbb-backend:latest
docker run -v .:/run/secrets -p 8000:8000 pyrallux/ccbb-backend:latest

Webserver:
docker build --secret id=generated-private-key,src="../ssl-private-key.txt" --secret id=generated-public-key,src="../ssl-public-key.crt" -t pyrallux/ccbb-webserver:latest .
docker push pyrallux/ccbb-webserver:latest
